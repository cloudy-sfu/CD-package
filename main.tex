% PAGE 1: FRONT SIDE (OUTSIDE OF SLEEVE)
% ==========================================
\begin{tikzpicture}[overlay, remember picture]
    % Origin is at the bottom-left of the A4 page
    \coordinate (BL) at (current page.south west);

    % ------------------------------------------
    % Draw Fold Lines (Dashed)
    % ------------------------------------------
    \draw[dashed, thick, gray] ([xshift=4.5cm]BL) -- ++(0, 24);
    \draw[dashed, thick, gray] ([xshift=16.5cm]BL) -- ++(0, 24);
    \draw[dashed, thick, gray] ([yshift=12.0cm]BL) -- ++(21, 0);
    \draw[dashed, thick, gray] ([xshift=4.5cm, yshift=24.0cm]BL) -- ++(12, 0);

    % ------------------------------------------
    % Draw Cut Lines (Solid, Black)
    % ------------------------------------------
    \draw[dashed, thick, gray] ([yshift=24cm]BL) -- ([xshift=4.5cm, yshift=24cm]BL) 
        -- ([xshift=7.35cm, yshift=29.7cm]BL) -- ([xshift=13.65cm, yshift=29.7cm]BL) 
        -- ([xshift=16.5cm, yshift=24cm]BL) -- ([xshift=21cm, yshift=24cm]BL);

    % ------------------------------------------
    % 1. Front Cover (Center 12cm x 12cm)
    % ------------------------------------------
    \begin{scope}[shift={($(BL)+(10.5cm, 16.5cm)$)}]
        \newcommand{\AlbumCoverPlaceholder}{
            \draw[thick] (-5, -3.0) rectangle (5, 7.0);
            \draw[dashed, gray] (-5, -3.0) -- (5, 7.0);
            \draw[dashed, gray] (-5, 7.0) -- (5, -3.0);
            \node[fill=white, align=center] at (0, 2.0) 
            {\normalsize \textbf{\AlbumCoverText}};
        }

        \ifx\AlbumCoverPath\empty
            \AlbumCoverPlaceholder
        \else
            \IfFileExists{\AlbumCoverPath}{
                \node[inner sep=0pt] at (0, 2.0) {\includegraphics[width=10cm, height=10cm]{\AlbumCoverPath}};
            }{
                \AlbumCoverPlaceholder
            }
        \fi

        \ifx\AlbumName\empty
            \node[anchor=west] at (-5.0, -3.5) {\normalsize \AlbumNameText};
        \else
            \node[anchor=west] at (-5.0, -3.5) {\normalsize \AlbumName};
        \fi
        
        \ifx\AlbumAuthors\empty
            \node[anchor=west] at (-5.0, -4.0) {\normalsize \AlbumAuthorsText};
        \else
            \node[anchor=west] at (-5.0, -4.0) {\normalsize \AlbumAuthors};
        \fi
    \end{scope}

    % ------------------------------------------
    % 2. Bottom Panel (Back Cover - Tracklist)
    % Using 'transform shape' correctly rotates the text content
    % ------------------------------------------
    \begin{scope}[shift={($(BL)+(10.5cm, 6.0cm)$)}, rotate=180, transform shape]
        \node[align=center] at (0, 4.2) {\Large \textbf{\TrackListText}};
        \draw[thick] (-4.5, -4.5) rectangle (4.5, 3.2);
        
        \ifx\TrackList\empty
            % Generate 4 track lines
            \foreach \y [count=\i] in {1.8, 0.3, -1.2, -2.7} {
                \node[anchor=west] at (-4, \y) {\textbf{0\i}};
                \draw[thick] (-3.2, \y - 0.15) -- (4, \y - 0.15);
            }
        \else
            % Configuration controls (defaults; can be overridden in config.tex)
            \providecommand{\TrackCols}{2}          % Set 1 for single column, 2 for two columns
            \providecommand{\TrackStep}{0.48}       % Vertical row gap in cm (e.g., 0.45 to 0.70)
            \providecommand{\TrackFontSize}{\small} % \normalsize, \small, or \footnotesize
            % Calculate how many rows fit inside the ~6.8cm vertical box height
            \pgfmathtruncatemacro{\TracksPerCol}{floor(6.8 / \TrackStep)}
            \pgfmathtruncatemacro{\MaxTracks}{\TracksPerCol * \TrackCols}
            \begin{scope}
                \clip (-4.5, -4.5) rectangle (4.5, 3.2);
                \foreach \Track [count=\i] in \TrackList {
                    % Compute row and column coordinates
                    \ifnum\TrackCols=2
                        \pgfmathtruncatemacro{\col}{(\i-1) / \TracksPerCol}
                        \pgfmathtruncatemacro{\row}{mod(\i-1, \TracksPerCol)}
                        \pgfmathsetmacro{\xnum}{-4.2 + \col * 4.4}
                        \pgfmathsetmacro{\xtext}{-3.5 + \col * 4.4}
                        \pgfmathsetmacro{\xend}{-0.2 + \col * 4.4}
                    \else
                        \pgfmathtruncatemacro{\col}{0}
                        \pgfmathtruncatemacro{\row}{\i-1}
                        \pgfmathsetmacro{\xnum}{-4.0}
                        \pgfmathsetmacro{\xtext}{-3.2}
                        \pgfmathsetmacro{\xend}{4.0}
                    \fi
                    \pgfmathsetmacro{\y}{2.6 - \row * \TrackStep}
                    % Track Number & Track Name
                    \node[anchor=west, font=\TrackFontSize] at (\xnum, \y) {\textbf{\ifnum\i<10 0\fi\i}};
                    \node[anchor=west, font=\TrackFontSize] at (\xtext, \y) {\Track};
                    \xdef\TotalTracks{\i}
                }
            \end{scope}
            \ifnum\TotalTracks>\MaxTracks
                \node[anchor=center, font=\TrackFontSize] at (0, -4.9) {\More ...};
            \fi
        \fi
    \end{scope}

    % ------------------------------------------
    % 3. Top Flap (Closing Flap)
    % ------------------------------------------
    \begin{scope}[shift={($(BL)+(10.5cm, 26.85cm)$)}, rotate=180, transform shape]
        \node[align=center] at (0, -0.6) {\large \textbf{\TopFoldI}};
        \node[align=center] at (0, -1.4) {\TopFoldII};
    \end{scope}
\end{tikzpicture}
